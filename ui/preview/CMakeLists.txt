cmake_minimum_required(VERSION 3.16)
project(UiPreview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 Xml Concurrent Gui Svg Test)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
qt_add_executable(ui-preview "${repo}/ui/preview/main.cpp" "${repo}/BackpackGenerator.cpp" "${repo}/BackpackGenerator.h" "${repo}/LocalizationManager.cpp" "${repo}/LocalizationManager.h")
target_include_directories(ui-preview PRIVATE "${repo}")
target_compile_definitions(ui-preview PRIVATE UI_SOURCE_DIR="${repo}")
target_link_libraries(ui-preview PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Xml Qt6::Concurrent Qt6::Gui Qt6::Svg Qt6::Test)
set_target_properties(ui-preview PROPERTIES WIN32_EXECUTABLE TRUE)
file(GLOB preview_svg_assets CONFIGURE_DEPENDS "${repo}/assets/*.svg")
qt_add_resources(ui-preview "preview_assets" PREFIX "/" BASE "${repo}" FILES "${repo}/backpack_templates.json" "${repo}/locales/zh_CN.json" "${repo}/locales/en_US.json" ${preview_svg_assets})
